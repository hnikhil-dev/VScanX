126def print_finding(finding: dict) -> None: 

127 """Print the stylized red vulnerability alert block.""" 

128 severity = str(finding.get("severity", "UNKNOWN")).upper() 

129 

130 # Silence generic INFO level status events from displaying as scary Vulnerability alerts 

131 if severity == "INFO": 

132 return 

133 

134 module = finding.get("module", "UNKNOWN") 

135 endpoint = finding.get("endpoint", "UNKNOWN") 

136 parameter = finding.get("parameter", "") 

137 evidence = finding.get("evidence") 

138 description = finding.get("description", "") 

139 

140 # Clean fallback for empty endpoint/URL 

141 if not endpoint or str(endpoint).strip() in ("", "None", "N/A"): 

142 endpoint = "N/A" 

143 

144 payload_str = "N/A" 

145 

146 # Check if a specific parameter is highlighted 

147 if parameter and str(parameter).strip() not in ("", "None", "N/A"): 

148 payload_str = f"Parameter: {parameter}" 

149 

150 # Check for evidence details or proof-of-concept payloads 

151 if evidence: 

152 if isinstance(evidence, dict): 

153 if "payload" in evidence and evidence["payload"]: 

154 payload_str = f"{evidence['payload']}" 

155 elif "trigger" in evidence and evidence["trigger"]: 

156 payload_str = f"Trigger: {evidence['trigger']}" 

157 elif "vector" in evidence and evidence["vector"]: 

158 payload_str = f"Vector: {evidence['vector']}" 

159 else: 

160 payload_str = ", ".join(f"{k}: {v}" for k, v in evidence.items() if v) 

161 else: 

162 payload_str = str(evidence) 

163 

164 # Include description if parameter/evidence is not detailed 

165 if payload_str == "N/A" and description: 

166 payload_str = description 

167 

168 # Style block matching Photoshop layout 

169 print(f"{RED}{BOLD}[!] Vulnerability Found{RESET}") 

170 print(f" {BOLD}Type :{RESET} {severity} {module}") 

171 print(f" {BOLD}URL :{RESET} {endpoint}") 

172 print(f" {BOLD}Payload :{RESET} {payload_str}\n")

183 print(f" {RED}HIGH: {by_sev.get('HIGH', 0)}{RESET}") 

184 print(f" {YELLOW}MEDIUM: {by_sev.get('MEDIUM', 0)}{RESET}") 

185 print(f" {GREEN}LOW: {by_sev.get('LOW', 0)}{RESET}") 

186 print(f" {WHITE}INFO: {by_sev.get('INFO', 0)}{RESET}") 

187 if summary.get("authenticated"): 

188 print(f" {CYAN}Authentication: ENABLED{RESET}") 

189 print(f"{CYAN}{BOLD}" + "=" * 60 + f"{RESET}")

191 if report_paths: 

192 print(f"\n{GREEN}{BOLD}[+] Reports Generated:{RESET}") 

193 for path in report_paths: 

194 print(f" * {path}")
